#include<stdio.h>
#include<conio.h>
#include<stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *head = 0;

void create()
{
    int n, i;
    struct node *newnode, *temp;
    printf("Enter number of nodes : ");
    scanf("%d", &n);
    for(i = 1; i <= n; i++)
    {
        newnode = (struct node *)malloc(sizeof(struct node));
        printf("Enter data : ");
        scanf("%d", &newnode->data);
        newnode->next = 0;
        if(head == 0)
        {
            head = temp = newnode;
        }
        else
        {
            temp->next = newnode;
            temp = newnode;
        }
    }
}

void deleteBeginning()
{
    struct node *del;
    if(head == 0)
    {
        printf("\nList is Empty");
    }
    else
    {
        del = head;
        head = head->next;
        free(del);
    }
}

void deletePosition()
{
    int pos, i;
    struct node *temp, *del;
    printf("\nEnter Position : ");
    scanf("%d", &pos);
    if(head == 0)
    {
        printf("\nList is Empty");
        return;
    }
    if(pos == 1)
    {
        del = head;
        head = head->next;
        free(del);
    }
    else
    {
        temp = head;
        for(i = 1; i < pos - 1 && temp != 0; i++)
            temp = temp->next;
        if(temp == 0 || temp->next == 0)
        {
            printf("\nInvalid Position");
        }
        else
        {
            del = temp->next;
            temp->next = del->next;
            free(del);
        }
    }
}

void deleteEnd()
{
    struct node *temp, *del;
    if(head == 0)
    {
        printf("\nList is Empty");
    }
    else if(head->next == 0)
    {
        free(head);
        head = 0;
    }
    else
    {
        temp = head;
        while(temp->next->next != 0)
            temp = temp->next;
        del = temp->next;
        temp->next = 0;
        free(del);
    }
}

void display()
{
    struct node *temp;
    temp = head;
    printf("\nLinked List : ");
    while(temp != 0)
    {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }
    printf("NULL\n");
}

void main()
{
    clrscr();

    create();
    printf("\nOriginal Linked List:");
    display();

    deleteBeginning();
    printf("\nAfter deleting at beginning:");
    display();

    deletePosition();
    printf("\nAfter deleting at specified position:");
    display();

    deleteEnd();
    printf("\nAfter deleting at end:");
    display();

    printf("\nCOMPLETED");
    getch();
}
